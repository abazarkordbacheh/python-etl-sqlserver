Annotated ==> git tag -a v1.0.0 -m "Release: first stable version"

========================== Undo and Redo Changes ==========================
git restore <file>
git restore --staged <file> # Staging Area (Unstage)
git restore --staged --worktree <file> #  Staging

git revert 4f2a91b
git revert HEAD

git reset --soft HEAD~1 # Staged
git reset --mixed HEAD~1 # Unstaged
git reset --hard HEAD~1
